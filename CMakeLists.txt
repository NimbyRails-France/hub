cmake_minimum_required(VERSION 3.24)
project(NRFHub VERSION 0.3.1 LANGUAGES CXX)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" NRF_RELEASE_VERSION LIMIT_COUNT 1)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Network)
qt_standard_project_setup()
qt_add_executable(NRFHub WIN32 src/main.cpp src/hub.cpp src/hub-desktop.cpp src/hub.h src/updater.cpp src/updater.h src/releases.cpp src/releases.h)
target_compile_features(NRFHub PRIVATE cxx_std_20)
target_compile_definitions(NRFHub PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX HUB_VERSION="${NRF_RELEASE_VERSION}")
target_link_libraries(NRFHub PRIVATE Qt6::Widgets Qt6::Network)
option(NRF_DEPLOY_QT "Deploy the Qt runtime for local Run/Debug" ON)
if(WIN32 AND NRF_DEPLOY_QT)
    find_program(NRF_WINDEPLOYQT windeployqt HINTS "${Qt6_DIR}/../../../bin" REQUIRED)
    add_custom_command(TARGET NRFHub POST_BUILD
        COMMAND "${NRF_WINDEPLOYQT}" --release --no-translations "$<TARGET_FILE:NRFHub>"
        VERBATIM)
endif()
add_custom_command(TARGET NRFHub POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/scripts" "$<TARGET_FILE_DIR:NRFHub>/scripts")
enable_testing()
add_test(NAME hub_self_test COMMAND NRFHub --self-test)
add_test(NAME hub_desktop_test COMMAND NRFHub --ui-test)
qt_add_executable(hub_release_tests tests/releases.cpp src/releases.cpp)
target_include_directories(hub_release_tests PRIVATE src)
target_link_libraries(hub_release_tests PRIVATE Qt6::Network)
target_compile_features(hub_release_tests PRIVATE cxx_std_20)
add_test(NAME release_channels COMMAND hub_release_tests)
